--- QUERY 2: Driver Available Pending Queue ---
SQL:
SELECT 
      id AS trip_id,
      pickup_address,
      destination_address,
      fare_amount_minor,
      currency,
      requested_at
    FROM trips
    WHERE status = 'REQUESTED' 
      AND driver_id IS NULL
    ORDER BY requested_at DESC
    LIMIT 10;
