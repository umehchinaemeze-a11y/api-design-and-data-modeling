SQL:
SELECT 
      t.id AS trip_id,
      t.status,
      t.fare_amount_minor,
      t.currency,
      t.driver_name_snapshot,
      t.vehicle_description_snapshot,
      t.pickup_address,
      t.destination_address,
      t.completed_at,
      p.status AS payment_status
    FROM trips t
    LEFT JOIN payments p ON p.trip_id = t.id
    WHERE t.rider_id = $1 
      AND t.status = 'COMPLETED'
    ORDER BY t.completed_at DESC
    LIMIT 5 OFFSET 0;
